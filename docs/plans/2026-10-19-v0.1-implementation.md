# dsh-vscode v0.1 Implementation Plan

> **For agentic workers:** Steps use checkbox (`- [ ]`) syntax for tracking. Execute task by task; commit at each checkpoint.

**Goal:** A VS Code extension whose `DSH: Open Panel` command opens the DeepSeek Harness web GUI in an editor panel, auto-starting `dsh web` when it is not running.

**Architecture:** Zero-dependency TypeScript extension (tsc-compiled CJS). `serverManager` probes/starts the existing `dsh web` server; `panel` embeds `http://127.0.0.1:<port>` in a webview iframe. No backend changes. Server response carries no `X-Frame-Options`/CSP `frame-ancestors`, and its root HTML contains `__DSH_BOOT__` (used as the health probe marker).

**Tech Stack:** VS Code Extension API (^1.85.0), TypeScript, tsc, @vscode/vsce.

**Testing:** Manual acceptance per spec §验收 (user-approved; automated tests are an explicit v0.1 non-goal).

```ts
/** Probe and lifecycle management for the local `dsh web` server. */
import { spawn, spawnSync, type ChildProcess } from 'node:child_process'
import { existsSync } from 'node:fs'
import * as vscode from 'vscode'

/** User-configured server location and behavior. */
export interface ServerConfig {
  repoPath: string
  port: number
  autoStart: boolean
}

/** Marker in the dsh web root HTML; distinguishes dsh from a foreign process on the port. */
const BOOT_MARKER = '__DSH_BOOT__'
const START_TIMEOUT_MS = 60_000
const POLL_INTERVAL_MS = 1_000

type Probe = 'dsh' | 'foreign' | 'free'

let child: ChildProcess | undefined
let output: vscode.OutputChannel | undefined

/** Classify what answers on `port`: dsh web, a foreign HTTP process, or nothing. */
async function probe(port: number): Promise<Probe> {
  try {
    const res = await fetch(`http://127.0.0.1:${port}/`, { signal: AbortSignal.timeout(2_000) })
    if (!res.ok) return 'foreign'
    return (await res.text()).includes(BOOT_MARKER) ? 'dsh' : 'foreign'
  } catch {
    return 'free'
  }
}

/** Resolve a node binary; VS Code's GUI-launch PATH may lack Homebrew. */
function resolveNode(): string {
  for (const candidate of ['node', '/opt/homebrew/bin/node', '/usr/local/bin/node']) {
    const r = spawnSync(candidate, ['--version'], { encoding: 'utf8' })
    if (r.status === 0) return candidate === 'node' ? 'node' : candidate
  }
  throw new Error('No node binary found (tried PATH, /opt/homebrew/bin, /usr/local/bin).')
}

/**
 * Return once a dsh web server answers on `cfg.port`, starting it when needed.
 * Throws with a user-facing message on foreign port occupation, missing repoPath,
 * early server exit, or startup timeout.
 */
export async function ensureRunning(cfg: ServerConfig): Promise<void> {
  const state = await probe(cfg.port)
  if (state === 'dsh') return
  if (state === 'foreign') {
    throw new Error(`Port ${cfg.port} is used by another application; change the dshHarness.port setting.`)
  }
  if (!cfg.autoStart) {
    throw new Error('dsh web is not running and dshHarness.autoStart is off.')
  }
  if (!existsSync(cfg.repoPath)) {
    throw new Error(`dshHarness.repoPath does not exist: ${cfg.repoPath}`)
  }
  const node = resolveNode()
  output ??= vscode.window.createOutputChannel('DeepSeek Harness Server')
  output.appendLine(`[dsh] start: ${node} --import tsx/esm apps/cli/src/bin.ts web (cwd: ${cfg.repoPath})`)
  child = spawn(node, ['--import', 'tsx/esm', 'apps/cli/src/bin.ts', 'web'], {
    cwd: cfg.repoPath,
    env: process.env,
  })
  child.stdout?.on('data', (d: Buffer) => output?.append(d.toString()))
  child.stderr?.on('data', (d: Buffer) => output?.append(d.toString()))
  child.on('exit', code => output?.appendLine(`[dsh] server exited (code ${code ?? 'null'})`))
  const deadline = Date.now() + START_TIMEOUT_MS
  while (Date.now() < deadline) {
    await new Promise(r => setTimeout(r, POLL_INTERVAL_MS))
    if (child.exitCode !== null) {
      output.show()
      throw new Error('dsh web exited during startup; see the "DeepSeek Harness Server" output channel.')
    }
    if ((await probe(cfg.port)) === 'dsh') return
  }
  output.show()
  throw new Error('Timed out (60s) waiting for dsh web; see the "DeepSeek Harness Server" output channel.')
}
```

/** Webview panel hosting the dsh web GUI. */
import * as vscode from 'vscode'

let panel: vscode.WebviewPanel | undefined

/** Create the panel on first use, reveal it afterwards. */
export function createOrReveal(port: number): void {
  if (panel) {
    panel.reveal()
    return
  }
  const url = `http://127.0.0.1:${port}`
  panel = vscode.window.createWebviewPanel('dshHarness', 'DeepSeek Harness', vscode.ViewColumn.Beside, {
    enableScripts: true,
    retainContextWhenHidden: true,
  })
  panel.webview.html = render(url)
  panel.onDidDispose(() => {
    panel = undefined
  })
}

/** CSP locks the host page down; the GUI itself loads inside the iframe. */
function render(url: string): string {
  return `<!doctype html>
<html><head>
<meta http-equiv="Content-Security-Policy" content="default-src 'none'; frame-src ${url}; style-src 'unsafe-inline';">
<style>html,body{margin:0;padding:0;height:100%;overflow:hidden}iframe{border:0;width:100%;height:100%}</style>
</head><body><iframe src="${url}"></iframe></body></html>`
}
```

/** Extension entry: register `DSH: Open Panel`. */
import * as vscode from 'vscode'
import { ensureRunning, type ServerConfig } from './serverManager'
import { createOrReveal } from './panel'

export function activate(context: vscode.ExtensionContext): void {
  context.subscriptions.push(
    vscode.commands.registerCommand('dsh-vscode.openPanel', async () => {
      const cfg = readConfig()
      try {
        await vscode.window.withProgress(
          { location: vscode.ProgressLocation.Notification, title: 'DSH: starting server…', cancellable: false },
          () => ensureRunning(cfg),
        )
      } catch (err) {
        vscode.window.showErrorMessage(`DSH: ${err instanceof Error ? err.message : String(err)}`)
        return
      }
      createOrReveal(cfg.port)
    }),
  )
}

function readConfig(): ServerConfig {
  const c = vscode.workspace.getConfiguration('dshHarness')
  return {
    repoPath: c.get('repoPath', '/Users/baojian/Desktop/deepseek-harness-demo/deepseek-harness'),
    port: c.get('port', 3080),
    autoStart: c.get('autoStart', true),
  }
}

/** The dsh web server keeps its own lifecycle; nothing to clean up here. */
export function deactivate(): void {}
```
